from core.db import get_connection

def listar():
    conn = get_connection()
    cur = conn.cursor()
    cur.execute("""
        SELECT C.CODCTA, C.DESCRCTA, A.CODCTAAGR, A.DESCRCTAAGR,
               T.TPCUSTO, C.NATUREZACTA, C.DTAABERTURACTA
        FROM CADCONTA C
        LEFT JOIN CADCTAAGR A ON C.CODCTAAGR = A.CODCTAAGR 
        LEFT JOIN CONTASPORCUSTO T ON C.TPCUSTO = T.TPCUSTO
        ORDER BY C.CODCTA
    """)
    
    return cur.fetchall()

def existe(cod_conta):
    conn = get_connection()
    cur = conn.cursor()
    cur.execute("SELECT COUNT(*) FROM CADCONTA WHERE CODCTA=?", (cod_conta,))
    return cur.fetchone()[0] > 0

def inserir(cod_conta, descricao, cod_agr, cod_custo, natureza, dtaabert):
    conn = get_connection()
    cur = conn.cursor()
    cur.execute("""
        INSERT INTO CADCONTA (CODCTA, DESCRCTA, CODCTAAGR, TPCUSTO, NATUREZACTA, DTAABERTURACTA)
        VALUES (?, ?, ?, ?, ?, ?)
    """, (cod_conta, descricao, cod_agr, cod_custo, natureza, dtaabert))
    conn.commit()

def atualizar(cod_conta, descricao, cod_agr, cod_custo, natureza, dtaabert):
    conn = get_connection()
    cur = conn.cursor()
    cur.execute("""
        UPDATE CADCONTA
        SET DESCRCTA=?, CODCTAAGR=?, TPCUSTO=?, NATUREZACTA=?, DTAABERTURACTA=?
        WHERE CODCTA=?
    """, (descricao, cod_agr, cod_custo, natureza, dtaabert, cod_conta))
    conn.commit()

def excluir(cod_conta):
    conn = get_connection()
    cur = conn.cursor()
    cur.execute("DELETE FROM CADCONTA WHERE CODCTA=?", (cod_conta,))
    conn.commit()

import tkinter as tk
from tkinter import messagebox, ttk
import sql.sql_contaspocusto as sql

def tela_contaspocusto():

    # Guarda o registro original selecionado no Treeview
    registro_original = {"TPCUSTO": None, "CONTA": None}

    def atualizar_lista():
        for item in tree.get_children():
            tree.delete(item)

        for r in sql.listar():
            # Inserção limpa no Treeview (sem aspas, sem parênteses)
            tree.insert("", tk.END, values=(r[0], r[1], r[2]))

    def preencher_campos(event):
        item = tree.selection()
        if item:
            valores = tree.item(item, "values")

            # Preenche os campos
            entry_tpCusto.delete(0, tk.END)
            entry_tpCusto.insert(0, valores[0])

            entry_Conta.delete(0, tk.END)
            entry_Conta.insert(0, valores[1])

            entry_percentual.delete(0, tk.END)
            entry_percentual.insert(0, valores[2])

            # Guarda o registro original para validação futura
            registro_original["TPCUSTO"] = valores[0]
            registro_original["CONTA"] = valores[1]

    def inserir():
        if not messagebox.askyesno("Confirmação", "Deseja inserir este registro?"):
            return

        tp = entry_tpCusto.get()
        conta = entry_Conta.get()
        perc = entry_percentual.get()

        if sql.existe(tp, conta):
            messagebox.showwarning("Duplicado", "Este registro já existe!")
            return

        ok = sql.inserir(tp, conta, perc)
        if ok:
            atualizar_lista()
            messagebox.showinfo("OK", "Tipo de custo inserido.")
        else:
            messagebox.showwarning("Erro", "Não foi possível inserir.")

    def atualizar():
        if not messagebox.askyesno("Confirmação", "Deseja atualizar este registro?"):
            return

        tp_original = registro_original["TPCUSTO"]
        conta_original = registro_original["CONTA"]

        # Nenhum registro selecionado antes
        if tp_original is None or conta_original is None:
            messagebox.showwarning("Erro", "Selecione um registro antes de atualizar.")
            return

        # Verifica se o registro original ainda existe no banco
        if not sql.existe(tp_original, conta_original):
            messagebox.showerror("Inexistente", "O registro original não existe mais no cadastro.")
            return

        # Novos valores digitados pelo usuário
        tp_novo = entry_tpCusto.get()
        conta_novo = entry_Conta.get()
        perc_novo = entry_percentual.get()

        # Se o usuário mudou a chave primária, verificar duplicidade
        if (tp_novo != tp_original or conta_novo != conta_original) and sql.existe(tp_novo, conta_novo):
            messagebox.showwarning("Duplicado", "Já existe um registro com esta nova chave.")
            return

        # Atualiza corretamente
        sql.excluir(tp_original, conta_original)  # remove o antigo
        sql.inserir(tp_novo, conta_novo, perc_novo)  # insere o novo

        atualizar_lista()
        messagebox.showinfo("OK", "Registro atualizado com sucesso.")

        # Limpa registro original
        registro_original["TPCUSTO"] = None
        registro_original["CONTA"] = None

    def excluir():
        if not messagebox.askyesno("Confirmação", "Deseja excluir este registro?"):
            return

        tp = entry_tpCusto.get()
        conta = entry_Conta.get()

        if not sql.existe(tp, conta):
            messagebox.showerror("Inexistente", "Este registro não existe.")
            return

        sql.excluir(tp, conta)
        atualizar_lista()
        messagebox.showinfo("OK", "Tipo de custo excluído.")

    # Construção da janela
    janela = tk.Toplevel()
    janela.title("Cadastro de Tipos de Custo")
    janela.geometry("600x600")

    tk.Label(janela, text="Código Tipo de Custo").pack()
    entry_tpCusto = tk.Entry(janela)
    entry_tpCusto.pack()

    tk.Label(janela, text="Conta Contábil envolvida").pack()
    entry_Conta = tk.Entry(janela)
    entry_Conta.pack()

    tk.Label(janela, text="Percentual").pack()
    entry_percentual = tk.Entry(janela)
    entry_percentual.pack()

    tk.Button(janela, text="Inserir", command=inserir).pack(pady=5)
    tk.Button(janela, text="Atualizar", command=atualizar).pack(pady=5)
    tk.Button(janela, text="Excluir", command=excluir).pack(pady=5)

    # TREEVIEW (spreadsheet-like)
    colunas = ("TPCUSTO", "CONTA", "PERCENTUAL")
    tree = ttk.Treeview(janela, columns=colunas, show="headings", height=15)

    for col in colunas:
        tree.heading(col, text=col)
        tree.column(col, width=150)

    tree.pack(pady=10)
    tree.bind("<<TreeviewSelect>>", preencher_campos)

    atualizar_lista()

import tkinter as tk
from telas.tela_cadctaagr import tela_cadctaagr
from telas.tela_contaspocusto import tela_contaspocusto
from telas.tela_cadconta import tela_cadconta

def main():
    root = tk.Tk()
    root.title("Sistema de Cadastros")
    root.geometry("600x400")

    # === MENU PRINCIPAL ===
    menubar = tk.Menu(root)

    # === MENU CADASTRO ===
    menu_cadastro = tk.Menu(menubar, tearoff=0)
    menu_cadastro.add_command(label="Cadastro de Grupos", command=tela_cadctaagr)
    menu_cadastro.add_command(label="Cadastro de Tipos de Custo", command=tela_contaspocusto)
    menu_cadastro.add_command(label="Cadastro de Contas", command=tela_cadconta)

    menubar.add_cascade(label="Cadastro", menu=menu_cadastro)

    # === MENU FINALIZAR ===
    menu_finalizar = tk.Menu(menubar, tearoff=0)
    menu_finalizar.add_command(label="Sair do Sistema", command=root.quit)

    menubar.add_cascade(label="Finalizar", menu=menu_finalizar)

    # Aplica o menu na janela
    root.config(menu=menubar)

    # Mensagem inicial ou área futura para dashboard
    label = tk.Label(root, text="Bem-vindo ao Sistema de Cadastros", font=("Arial", 16))
    label.pack(pady=40)

    root.mainloop()

if __name__ == "__main__":
    main()
